"""### Build the Recommender
In ALS, users and products (movies) are described by a small set of latent features (factors) that can be used to predict missing entries.

#### Latent Features
latent features might be things like properties and genre of the movies: action, romance, comedy...
We can use the features to produce some sort of algorithm (**ALS**) to intelligently calculate ratings

The ALS class is an estimator, so you can use its **fit** method to traing a model, or you can include it in a pipeline. Rather than specifying a feature vector and as label, the ALS algorithm requries a numeric user ID, item ID, and rating.
"""

# Commented out IPython magic to ensure Python compatibility.

# %pyspark
from pyspark.sql import functions as F
from pyspark.ml import Pipeline
from pyspark.ml.feature import StringIndexer, OneHotEncoder, VectorAssembler
from pyspark.ml.classification import LogisticRegression
from pyspark.ml.tuning import ParamGridBuilder, TrainValidationSplit
from pyspark.ml.evaluation import (
    BinaryClassificationEvaluator,
    MulticlassClassificationEvaluator
)

# 1. Load the original complaints view.
# This uses ratings_csv, as shown in your traceback.
complaints = spark.table("ratings_csv")

# 2. Create the numeric target: Yes = 1, No = 0.
response = F.lower(F.trim(F.col("Timely response?")))

data = (
    complaints
    .filter(response.isin("yes", "no"))
    .withColumn(
        "label",
        F.when(response == "yes", 1.0).otherwise(0.0)
    )
    .select("Company", "Product", "Issue", "label")
    .fillna("Unknown", subset=["Company", "Product", "Issue"])
)

data.groupBy("label").count().show()

train, test = data.randomSplit([0.7, 0.3], seed=42)

# 3. Convert text categories into numeric features.
categories = ["Company", "Product", "Issue"]

indexers = [
    StringIndexer(
        inputCol=c,
        outputCol=c + "_index",
        handleInvalid="keep"
    )
    for c in categories
]

encoder = OneHotEncoder(
    inputCols=[c + "_index" for c in categories],
    outputCols=[c + "_vector" for c in categories]
)

assembler = VectorAssembler(
    inputCols=[c + "_vector" for c in categories],
    outputCol="features"
)

# 4. Use logistic regression instead of ALS.
lr = LogisticRegression(
    featuresCol="features",
    labelCol="label",
    maxIter=50
)

pipeline = Pipeline(
    stages=indexers + [encoder, assembler, lr]
)

# 5. Build a NEW parameter grid and validation object.
paramGrid = (
    ParamGridBuilder()
    .addGrid(lr.regParam, [0.01, 0.1])
    .build()
)

evaluator = BinaryClassificationEvaluator(
    labelCol="label",
    rawPredictionCol="rawPrediction",
    metricName="areaUnderROC"
)

cv = TrainValidationSplit(
    estimator=pipeline,
    estimatorParamMaps=paramGrid,
    evaluator=evaluator,
    trainRatio=0.8,
    seed=42
)

model = cv.fit(train)

# 6. Predict and evaluate on held-out test data.
prediction = model.transform(test)

prediction.select(
    "Company", "Product", "Issue", "label", "prediction"
).show(10, truncate=False)

print("Test ROC AUC:", evaluator.evaluate(prediction))

accuracy_evaluator = MulticlassClassificationEvaluator(
    labelCol="label",
    predictionCol="prediction",
    metricName="accuracy"
)

print("Test accuracy:", accuracy_evaluator.evaluate(prediction))

# Show which outcomes the model gets right or wrong.
prediction.groupBy("label", "prediction").count().show()

# Commented out IPython magic to ensure Python compatibility.

# %pyspark
from pyspark.ml import Pipeline
from pyspark.ml.feature import StringIndexer, OneHotEncoder, VectorAssembler
from pyspark.ml.classification import LogisticRegression
from pyspark.ml.tuning import ParamGridBuilder, TrainValidationSplit
from pyspark.ml.evaluation import BinaryClassificationEvaluator

columns = ["Company", "Product", "Issue"]

indexers = [
    StringIndexer(
        inputCol=c,
        outputCol=c + "_index",
        handleInvalid="keep"
    )
    for c in columns
]

encoder = OneHotEncoder(
    inputCols=[c + "_index" for c in columns],
    outputCols=[c + "_vector" for c in columns]
)

assembler = VectorAssembler(
    inputCols=[c + "_vector" for c in columns],
    outputCol="features"
)

classifier = LogisticRegression(
    labelCol="label",
    featuresCol="features",
    maxIter=50
)

classification_pipeline = Pipeline(
    stages=indexers + [encoder, assembler, classifier]
)

classification_grid = (
    ParamGridBuilder()
    .addGrid(classifier.regParam, [0.01, 0.1])
    .build()
)

classification_evaluator = BinaryClassificationEvaluator(
    labelCol="label",
    metricName="areaUnderROC"
)

cv = TrainValidationSplit(
    estimator=classification_pipeline,
    estimatorParamMaps=classification_grid,
    evaluator=classification_evaluator,
    trainRatio=0.8,
    seed=42
)

model = cv.fit(train)
prediction = model.transform(test)

prediction.select("label", "prediction", "probability").show(10)
print("Test ROC AUC:", classification_evaluator.evaluate(prediction))
